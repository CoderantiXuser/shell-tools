#!/bin/bash

# Advanced System Check Script for Networking Issues
# Focus: wlan0 auto-start problems and general networking health

LOG_FILE="/var/log/network_diagnostic_$(date +%Y%m%d_%H%M%S).log"
TEMP_DIR="/tmp/network_check_$(date +%s)"

# Create temp directory
mkdir -p "$TEMP_DIR"

# Color codes for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# Function to log and print messages
log() {
    echo -e "$(date '+%Y-%m-%d %H:%M:%S') - $1" | tee -a "$LOG_FILE"
}

# Header for the log file
echo "==================== NETWORK DIAGNOSTIC REPORT ====================" > "$LOG_FILE"
echo "Date: $(date)" >> "$LOG_FILE"
echo "Hostname: $(hostname)" >> "$LOG_FILE"
echo "================================================================" >> "$LOG_FILE"

# System Overview
log "${YELLOW}=== SYSTEM OVERVIEW ===${NC}"
log "Checking system information..."
log "Kernel: $(uname -r)"
log "OS: $(lsb_release -d | cut -f2-)"
log "Uptime: $(uptime)"
log "CPU: $(grep -m 1 'model name' /proc/cpuinfo | cut -d: -f2 | xargs)"
log "Memory: $(free -h | awk '/Mem/{print $3 "/" $2}') used"
log "Load Average: $(cat /proc/loadavg)"

# Network Interfaces Overview
log "\n${YELLOW}=== NETWORK INTERFACES OVERVIEW ===${NC}"
log "Listing all network interfaces:"
ip -br link show | tee -a "$LOG_FILE"

# Focus on wlan0
log "\n${YELLOW}=== WLAN0 SPECIFIC CHECKS ===${NC}"
if ip link show wlan0 &>/dev/null; then
    log "${GREEN}wlan0 interface exists${NC}"
    
    # Check if wlan0 is up
    if ip link show wlan0 | grep -q 'state UP'; then
        log "${GREEN}wlan0 is UP${NC}"
    else
        log "${RED}wlan0 is DOWN${NC}"
    fi
    
    # Check wlan0 driver info
    log "\nDriver information for wlan0:"
    ethtool -i wlan0 2>&1 | tee -a "$LOG_FILE"
    
    # Check wlan0 firmware
    log "\nFirmware information for wlan0:"
    dmesg | grep wlan0 | grep firmware | tail -5 | tee -a "$LOG_FILE"
    
    # Check wlan0 connection status
    log "\nConnection status for wlan0:"
    iw dev wlan0 link 2>&1 | tee -a "$LOG_FILE"
else
    log "${RED}wlan0 interface does NOT exist${NC}"
fi

# Network Manager Status
log "\n${YELLOW}=== NETWORK MANAGER STATUS ===${NC}"
log "Checking NetworkManager service..."
systemctl status NetworkManager --no-pager | tee -a "$LOG_FILE"

log "\nNetworkManager connections:"
nmcli connection show | tee -a "$LOG_FILE"

log "\nNetworkManager devices:"
nmcli device status | tee -a "$LOG_FILE"

if nmcli device | grep -q wlan0; then
    log "\nDetailed wlan0 information from NetworkManager:"
    nmcli device show wlan0 | tee -a "$LOG_FILE"
fi

# systemd-networkd Status
log "\n${YELLOW}=== SYSTEMD-NETWORKD STATUS ===${NC}"
log "Checking systemd-networkd service..."
systemctl status systemd-networkd --no-pager | tee -a "$LOG_FILE"

# wpa_supplicant Status
log "\n${YELLOW}=== WPA_SUPPLICANT STATUS ===${NC}"
log "Checking wpa_supplicant service..."
systemctl status wpa_supplicant --no-pager | tee -a "$LOG_FILE"

# DHCP Client Status
log "\n${YELLOW}=== DHCP CLIENT STATUS ===${NC}"
log "Checking DHCP client service..."
systemctl status dhcpcd --no-pager 2>&1 | tee -a "$LOG_FILE"
systemctl status dhclient --no-pager 2>&1 | tee -a "$LOG_FILE"

# Network Configuration Files
log "\n${YELLOW}=== NETWORK CONFIGURATION FILES ===${NC}"
log "Checking /etc/network/interfaces:"
if [ -f "/etc/network/interfaces" ]; then
    cat "/etc/network/interfaces" | tee -a "$LOG_FILE"
else
    log "File /etc/network/interfaces does not exist"
fi

log "\nChecking /etc/netplan/ files:"
if [ -d "/etc/netplan" ]; then
    ls -la /etc/netplan/ | tee -a "$LOG_FILE"
    for file in /etc/netplan/*.yaml; do
        log "\nContents of $file:"
        cat "$file" | tee -a "$LOG_FILE"
    done
else
    log "Directory /etc/netplan does not exist"
fi

log "\nChecking wpa_supplicant.conf:"
if [ -f "/etc/wpa_supplicant/wpa_supplicant.conf" ]; then
    cat "/etc/wpa_supplicant/wpa_supplicant.conf" | tee -a "$LOG_FILE"
else
    log "File /etc/wpa_supplicant/wpa_supplicant.conf does not exist"
fi

# Kernel Modules and Drivers
log "\n${YELLOW}=== KERNEL MODULES AND DRIVERS ===${NC}"
log "Checking loaded wireless modules:"
lsmod | grep -E 'wlan|wireless|80211|mac80211|cfg80211' | tee -a "$LOG_FILE"

log "\nChecking available wireless modules:"
find /lib/modules/$(uname -r)/ -name '*wireless*' -o -name '*wlan*' -o -name '*80211*' | tee -a "$LOG_FILE"

# Hardware Information
log "\n${YELLOW}=== HARDWARE INFORMATION ===${NC}"
log "Checking PCI wireless devices:"
lspci -nnk | grep -iA3 'network\|wireless' | tee -a "$LOG_FILE"

log "\nChecking USB wireless devices:"
lsusb | tee -a "$LOG_FILE"

# System Logs
log "\n${YELLOW}=== SYSTEM LOGS ===${NC}"
log "Checking journalctl for NetworkManager errors:"
journalctl -u NetworkManager --no-pager -n 50 | grep -i error | tee -a "$LOG_FILE"

log "\nChecking journalctl for wpa_supplicant errors:"
journalctl -u wpa_supplicant --no-pager -n 50 | grep -i error | tee -a "$LOG_FILE"

log "\nChecking dmesg for wireless errors:"
dmesg | grep -iE 'wlan|wireless|80211|firmware|error' | tail -50 | tee -a "$LOG_FILE"

# RF Kill Status
log "\n${YELLOW}=== RF KILL STATUS ===${NC}"
log "Checking rfkill list:"
rfkill list | tee -a "$LOG_FILE"

# Network Connections
log "\n${YELLOW}=== ACTIVE NETWORK CONNECTIONS ===${NC}"
log "Checking active connections:"
ss -tulnp | tee -a "$LOG_FILE"

# IP Configuration
log "\n${YELLOW}=== IP CONFIGURATION ===${NC}"
log "Checking ip addr:"
ip addr | tee -a "$LOG_FILE"

log "\nChecking ip route:"
ip route | tee -a "$LOG_FILE"

# DNS Configuration
log "\n${YELLOW}=== DNS CONFIGURATION ===${NC}"
log "Checking resolv.conf:"
cat /etc/resolv.conf | tee -a "$LOG_FILE"

log "\nChecking systemd-resolved status:"
systemctl status systemd-resolved --no-pager | tee -a "$LOG_FILE"

# Firewall Status
log "\n${YELLOW}=== FIREWALL STATUS ===${NC}"
log "Checking ufw status:"
ufw status verbose 2>&1 | tee -a "$LOG_FILE"

log "\nChecking iptables rules:"
iptables -L -n -v | tee -a "$LOG_FILE"

# Potential Conflicts
log "\n${YELLOW}=== POTENTIAL CONFLICTS ===${NC}"
log "Checking for multiple network services running:"
ps aux | grep -E 'NetworkManager|wpa_supplicant|dhclient|dhcpcd|systemd-networkd' | grep -v grep | tee -a "$LOG_FILE"

log "\nChecking for network-related udev rules:"
ls -la /etc/udev/rules.d/*net*.rules 2>/dev/null | tee -a "$LOG_FILE"
ls -la /etc/udev/rules.d/*wlan*.rules 2>/dev/null | tee -a "$LOG_FILE"

# Summary of Findings
log "\n${YELLOW}=== SUMMARY OF FINDINGS ===${NC}"

# Check if wlan0 exists
if ! ip link show wlan0 &>/dev/null; then
    log "${RED}CRITICAL: wlan0 interface not detected${NC}"
fi

# Check if wlan0 is up
if ip link show wlan0 2>/dev/null | grep -q 'state DOWN'; then
    log "${RED}WARNING: wlan0 interface is DOWN${NC}"
fi

# Check for NetworkManager errors
if journalctl -u NetworkManager --no-pager -n 50 | grep -qi 'error'; then
    log "${RED}WARNING: NetworkManager has reported errors${NC}"
fi

# Check for wpa_supplicant errors
if journalctl -u wpa_supplicant --no-pager -n 50 | grep -qi 'error'; then
    log "${RED}WARNING: wpa_supplicant has reported errors${NC}"
fi

# Check for rfkill blocking
if rfkill list | grep -qi 'yes'; then
    log "${RED}WARNING: Some wireless devices are soft-blocked${NC}"
fi

# Check for multiple DHCP clients
dhcp_clients=$(ps aux | grep -E 'dhclient|dhcpcd' | grep -v grep | wc -l)
if [ "$dhcp_clients" -gt 1 ]; then
    log "${RED}WARNING: Multiple DHCP clients running (potential conflict)${NC}"
fi

# Final message
log "\n${GREEN}Diagnostic check completed. Full report saved to $LOG_FILE${NC}"
log "Please review the output above and check the log file for detailed information."

# Compress collected data for further analysis
tar czf "$TEMP_DIR.tar.gz" "$TEMP_DIR" 2>/dev/null
log "Additional system data collected in: $TEMP_DIR.tar.gz"

exit 0
